#Make a function that returns mean, median and sd for quantitative variables 

mydescr<-function(data){
mesos<-mean(data)
diamesos<-median(data)
sd<-sqrt(var(data))
return( list("mean"=mesos,"median"=diamesos,"sd"=sd) )
}
#Check that function is ok
mydescr(pizza_num[,1])
mydescr(pizza_num[,2])

#Apply to all variables
