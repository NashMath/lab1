# Define a function that returns the barplot or the histogram of a variable according to its type
myPlot <- function(data) {
  if (is.numeric(data)) {
    p <- hist(data)
  } else {
    p <- barplot(prop.table(table(data)))
  }
  return(p)
}


# Apply the function for the variable of price and Product.Category
myPlot(zara$price)
myPlot(zara$Product.Category)


#Remove the Product.Category from the dataset
zara<-zara[,-3]

# Apply the function for all variables
par(mfrow=c(2,3))
sapply(zara,myPlot)
##OR
for(i in 1:ncol(zara)){
myPlot(zara[,i])
}


##Define the main of the plot
myPlot2 <- function(data,nam) {
  if (is.numeric(data)) {
    p <- hist(data,main=nam)
  } else {
    p <- barplot(prop.table(table(data)),main=nam)
  }
  return(p)
}

par(mfrow=c(2,3))
for(i in 1:ncol(zara)){
myPlot2(zara[,i],names(zara)[i])
}

# Define a function that returns appropriate statistics of a variable according to its type
myDescr <- function(data) {
  if (is.numeric(data)) {
    p <- list(mean(data),median(data),sd(data))
  } else {
    p <- list(table(data),prop.table(table(data)))
  }
  return(p)
}

sapply(zara,myDescr)
